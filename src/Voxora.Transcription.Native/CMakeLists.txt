cmake_minimum_required(VERSION 3.21)
project(SineFlowWhisperNative LANGUAGES C CXX)

option(SINEFLOW_ENABLE_VULKAN "Build whisper.cpp with Vulkan support" OFF)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_IS_DEV OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
if(SINEFLOW_ENABLE_VULKAN)
    set(GGML_VULKAN ON CACHE BOOL "" FORCE)
endif()

add_subdirectory(${CMAKE_CURRENT_LIST_DIR}/../../third_party/whisper.cpp ${CMAKE_CURRENT_BINARY_DIR}/whisper.cpp)
add_library(SineFlowWhisperNative SHARED SineFlowWhisperNative.cpp)
target_compile_features(SineFlowWhisperNative PRIVATE cxx_std_20)
target_compile_definitions(SineFlowWhisperNative PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_include_directories(SineFlowWhisperNative PRIVATE ${CMAKE_CURRENT_LIST_DIR}/../../third_party/whisper.cpp/include)
target_link_libraries(SineFlowWhisperNative PRIVATE whisper)
set_target_properties(SineFlowWhisperNative PROPERTIES OUTPUT_NAME "SineFlowWhisperNative")
foreach(dependency whisper ggml ggml-base ggml-cpu)
    add_custom_command(TARGET SineFlowWhisperNative POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:${dependency}> $<TARGET_FILE_DIR:SineFlowWhisperNative>)
endforeach()
