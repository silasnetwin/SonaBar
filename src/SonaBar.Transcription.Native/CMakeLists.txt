cmake_minimum_required(VERSION 3.21)
project(SonaBarWhisperNative LANGUAGES C CXX)

option(SONABAR_ENABLE_VULKAN "Build whisper.cpp with Vulkan support" OFF)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_IS_DEV OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
if(SONABAR_ENABLE_VULKAN)
    set(GGML_VULKAN ON CACHE BOOL "" FORCE)
endif()

add_subdirectory(${CMAKE_CURRENT_LIST_DIR}/../../third_party/whisper.cpp ${CMAKE_CURRENT_BINARY_DIR}/whisper.cpp)
add_library(SonaBarWhisperNative SHARED SonaBarWhisperNative.cpp)
target_compile_features(SonaBarWhisperNative PRIVATE cxx_std_20)
target_compile_definitions(SonaBarWhisperNative PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_include_directories(SonaBarWhisperNative PRIVATE ${CMAKE_CURRENT_LIST_DIR}/../../third_party/whisper.cpp/include)
target_link_libraries(SonaBarWhisperNative PRIVATE whisper)
set_target_properties(SonaBarWhisperNative PROPERTIES OUTPUT_NAME "SonaBarWhisperNative")
foreach(dependency whisper ggml ggml-base ggml-cpu)
    add_custom_command(TARGET SonaBarWhisperNative POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:${dependency}> $<TARGET_FILE_DIR:SonaBarWhisperNative>)
endforeach()
